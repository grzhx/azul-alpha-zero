cmake_minimum_required(VERSION 3.20)
project(azul VERSION 1.0.0 LANGUAGES CXX)
option(AZUL_NATIVE "Optimize for the build machine (nonportable binaries)" OFF)
option(AZUL_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
enable_testing()
add_library(azul_core INTERFACE)
target_include_directories(azul_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(azul_core INTERFACE /W4 /permissive- /utf-8)
else()
  target_compile_options(azul_core INTERFACE -Wall -Wextra -Wpedantic)
  if(AZUL_NATIVE)
    target_compile_options(azul_core INTERFACE -march=native)
  endif()
  if(AZUL_SANITIZE)
    target_compile_options(azul_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(azul_core INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_library(azul SHARED src/c_api.cpp)
target_link_libraries(azul PRIVATE azul_core Threads::Threads)
target_compile_definitions(azul PRIVATE AZUL_BUILD_DLL)
add_executable(azul_bench src/benchmark.cpp)
target_link_libraries(azul_bench PRIVATE azul_core Threads::Threads)
add_executable(azul_tests tests/test_engine.cpp)
add_executable(azul_tests3 tests/test_engine3.cpp)
if(MSVC)
  target_compile_options(azul_tests3 PRIVATE /UNDEBUG)
else()
  target_compile_options(azul_tests3 PRIVATE -UNDEBUG)
endif()
target_link_libraries(azul_tests3 PRIVATE azul_core Threads::Threads azul)
add_test(NAME azul_tests3 COMMAND azul_tests3)
target_link_libraries(azul_tests PRIVATE azul_core Threads::Threads azul)
add_executable(azul_batch_bench src/batch_benchmark.cpp)
target_link_libraries(azul_batch_bench PRIVATE azul_core Threads::Threads azul)
add_test(NAME azul_tests COMMAND azul_tests)
add_executable(azul_search_tests tests/test_search.cpp)
target_link_libraries(azul_search_tests PRIVATE azul_core Threads::Threads azul)
add_test(NAME azul_search_tests COMMAND azul_search_tests)
